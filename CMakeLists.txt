cmake_minimum_required(VERSION 3.24)
project(recomp_image_core VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF)
if(NOT TARGET chdr-static)
  FetchContent_Declare(recomp_image_chdr
    URL "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libchdr-6cde5348eb118da3baf94f75a69577a005a484fd.tar.gz"
    URL_HASH SHA256=a222b1f1c842f988c2aad51e5f56ba94a423ed87ccab71f1abdb757caa7444f5
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(recomp_image_chdr)
endif()
add_library(recomp_image_core STATIC src/image_reader.cpp src/cue_sheet.cpp src/image_api.cpp src/image_sha256.c)
add_library(recomp-image-core::core ALIAS recomp_image_core)
target_include_directories(recomp_image_core PUBLIC include PRIVATE src)
target_compile_features(recomp_image_core PUBLIC cxx_std_17)
target_link_libraries(recomp_image_core PRIVATE chdr-static)
if(MSVC)
  target_compile_options(recomp_image_core PRIVATE /W4)
else()
  target_compile_options(recomp_image_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  foreach(scenario geometry cdda pregap sbi contract)
    add_executable(image_${scenario}_test tests/${scenario}.cpp)
    target_link_libraries(image_${scenario}_test PRIVATE recomp_image_core)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${scenario}")
    add_test(NAME image_${scenario} COMMAND image_${scenario}_test)
    set_tests_properties(image_${scenario} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${scenario}")
  endforeach()
  add_executable(image_c_api_test tests/c_api.c)
  target_link_libraries(image_c_api_test PRIVATE recomp_image_core)
  add_test(NAME image_c_api COMMAND image_c_api_test)
endif()
